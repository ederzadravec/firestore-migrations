/**
 * Firestore Adapter Implementation
 *
 * This is an example implementation for Firebase Admin SDK
 * Adapt this to your specific Firestore setup
 */

import * as admin from 'firebase-admin'
import { IFirestoreAdapter } from '../lib-migrations'

class FirestoreAdapter implements IFirestoreAdapter {
  async getItems<T>(collection: string): Promise<Array<T & { id: string }>> {
    const snapshot = await admin.firestore().collection(collection).get()
    return snapshot.docs.map(doc => ({
      ...(doc.data() as T),
      id: doc.id,
    }))
  }

  async getItemById<T>(collection: string, id: string): Promise<T & { id: string }> {
    const doc = await admin.firestore().collection(collection).doc(id).get()

    if (!doc.exists) {
      throw new Error(`Document not found: ${collection}/${id}`)
    }

    return {
      ...(doc.data() as T),
      id: doc.id,
    }
  }

  async createItemWithId(collection: string, id: string, data: any): Promise<string> {
    await admin.firestore().collection(collection).doc(id).set(data)
    return id
  }

  async updateItemById(collection: string, id: string, data: any): Promise<void> {
    await admin.firestore().collection(collection).doc(id).update(data)
  }

  async deleteItemById(collection: string, id: string): Promise<any> {
    return await admin.firestore().collection(collection).doc(id).delete()
  }
}

export const firestoreAdapter = new FirestoreAdapter()
